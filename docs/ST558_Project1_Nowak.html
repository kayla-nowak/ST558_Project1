Introduction

This project builds a set of generalized queries for accessing and summarizing data from the Public Use Microdata Sample (PUMS). PUMS is part of the US Census Bureau’s annual American Community Survey.

  • Data Source: https://www.census.gov/data/developers/data-sets/census-microdata-api.html
  • Variables (from 2024): https://api.census.gov/data/2024/acs/acs1/pums/variables.html

This project was designed to access the 1-Year PUMS data sets for years 2021 through 2024 only. Additionally, PUMS data for Puerto Rico are not included the scope of this project.

1. Data Processing

Variable Values

Pull and parse variable names and values for the PUMS data in a given year.

  • Input: year (2021, 2022, 2023, 2024)
  • Return: data/vars_yyyy.rda (as list)

Function:

getPUMSVars <- function(year){
  
  yearc <- as.character(year)

  ## Define variables of interest
  if (yearc %in% c("2021", "2022")) {
    want <- all_vars[-14]
  } else if (yearc %in% c("2023", "2024")) {
    want <- all_vars[-13]
  } else {
    stop("Invalid year input. Function only accepts 2021-2024")
  }
  
  ## Pull + Parse JSON
  url <- paste0("https://api.census.gov/data/",yearc,"/acs/acs1/pums/variables.json")
  pulled  <- GET(url)
  parsed  <- fromJSON(rawToChar(pulled$content))
  
  ## Subset List to Variables of Interest
  vars <- parsed$variables[want]
  
  ## Save to /data as .rds
  saveRDS(vars, file= paste0("data/vars",yearc,".rds"))
}

API Data Pull

The getPUMSData function …

  • year accepts 2021, 2022, 2023, 2024

  • nums accepts up to two numeric variables: PWGTP, AGEP, GASP, GRPIP, JWAP, JWDP, and/or JWMNP

    • if one variable is provided, “PWGTP” is the second variable
    • if two variables are provided, “PWGTP” must be one of the two variables
    • the default inputs are c(“AGEP”, “PWGTP”)

  • cats accepts one categorical variable: FER, HHL, SCH, SCHL, SEX.

    • the default input is “SEX”

  • geo accepts one level of geography: REGION, DIVISION, STATE

    • the default input is “STATE”
    • note: either “STATE” or “ST” may be used to request state-level geography

  • subgeo accepts one sub-level of geography:

    • Regions: Northeast, Puerto Rico, South, Midwest, West. Default = Northeast
    • Division: West South Central (South Region), Middle Atlantic (Northeast region), West North Central (Midwest region), East South Central (South region), East North Central (Midwest region), Pacific (West region), Puerto Rico, South Atlantic (South region), Mountain (West region), New England (Northeast region). Default = New England (Northeast region)
    • State: written as “State Name/XX” where XX is the two-letter abbreviation. Default = District of Columbia/DC
getPUMSData <- function(year, 
                        nums=c("AGEP","PWGTP"), 
                        cats=c("SEX"), 
                        geo=c("STATE"), 
                        subgeo=NULL, ...) {
  
  ## YEAR
  yearc <- as.character(year)
  
  #Check validity
  if (!yearc %in% c("2021", "2022", "2023", "2024")) {
    stop("Invalid year input. Function only accepts 2021-2024.")
  }
  
  ## NUMS
  nums <- toupper(nums)
  
  #Check validity
  if (length(nums) > 2 | (length(nums)==2 & !any(nums == "PWGTP")) ){
    stop("Invalid NUMS input. Only 1 numeric variable in addition to PWGTP may be requested.")
    
  } else if (any(!nums %in% c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP"))) {
    stop("Invalid NUMS input. The requested variable(s) are not on the approved numeric list.")
  } else if (length(nums==1) & nums == "PWGTP"){
    stop("Invalid NUMS input. At least 1 numeric variable other than PWGTP must be provided.")
  } else if (length(nums==1) & nums != "PWGTP"){
    #Automatically include PWGTP on a single-input request
    nums <- c(nums, "PWGTP")
  }
  
  ## CATS 
  cats <- toupper(cats)
  
  #Check validity
  if (length(cats) > 1){
    stop("Invalid CATS input. Only 1 categorical variable may be provided.")
    
  } else if (!cats %in% c("FER", "HHL", "SCH", "SCHL", "SEX")){
    stop("Invalid CATS input. The requested variable is not on the approved categorical list.")
    
  }

  ## GEO
  geo <- toupper(geo)
  
  #Check validity
  if (length(geo) > 1){
    stop("Invalid GEO input. Only 1 geography variable may be provided.")
    
  } else if (!geo %in% c("REGION", "DIVISION", "STATE", "ST")){
    stop("Invalid GEO input. The requested variable is not on the approved geography list.")
    
  } else {
    
    #Force "ST" vs. "STATE" if entered wrong.
    if(yearc %in% c("2021", "2022") & geo=="STATE") {
      geo <- "ST"
    } else if(yearc %in% c("2023", "2024") & geo=="ST") {
      geo <- "STATE"
    }
  }
  
  ## SUBGEO
  
  #Grab year-specific valid inputs
  vars <- readRDS(paste0("data/vars",yearc,".rds"))
  subR <- unlist(vars$REGION$values$item)
  subD <- unlist(vars$DIVISION$values$item)
  if(yearc %in% c("2021", "2022")) { 
    subS <- unlist(vars$ST$values$item)
  } else {
    subS <- unlist(vars$STATE$values$item) 
  }
  
  #Check validity
  if (length(subgeo) > 1){
    stop("Invalid SUBGEO input. Only 1 sub-geography variable may be provided.")
    
  } else if (geo=="REGION" & !subgeo %in% subR) {
    stop("Invalid SUBGEO input. The provided phrase is invalid for geo = REGION. Note: this field is case-sensitive.")
    
  } else if(geo=="DIVISION" & !subgeo %in% subD){
    stop("Invalid SUBGEO input. The provided phrase is invalid for geo = DIVISION Note: this field is case-sensitive.")
      
  } else if(geo %in% c("STATE", "ST") & !subgeo %in% subS){
    stop("Invalid SUBGEO input. The provided phrase is invalid for geo = STATE Note: this field is case-sensitive.")
      
  } else if (is.null(subgeo) & geo=="REGION") {
    subgeo <- "Northeast"  #set default 
    
  } else if(is.null(subgeo) & geo=="DIVISION") {
     subgeo <- "New England (Northeast region)"  #set default 
   
  } else if(is.null(subgeo) & geo %in% c("STATE", "ST")) {
     subgeo <- "District of Columbia/DC"  #set default 
   
  }
  
  
  ## Run API pull via helper function
  
}
