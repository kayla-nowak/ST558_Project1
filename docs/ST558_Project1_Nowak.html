Introduction

This project builds a set of generalized queries for accessing and summarizing data from the Public Use Microdata Sample (PUMS). PUMS is part of the US Census Bureau’s annual American Community Survey.

  • Data Source: https://www.census.gov/data/developers/data-sets/census-microdata-api.html
  • Variables (from 2024): https://api.census.gov/data/2024/acs/acs1/pums/variables.html

This project was designed to access the 1-Year PUMS data sets for years 2021 through 2024 only. Additionally, PUMS data for Puerto Rico are not included the scope of this project.

Variable Values

getPUMSVars() pulls and parses variable names and values for the PUMS data in a given year.

One year-specific .rda file is returned, and this file is saved in the repository’s /data folder for future access. Later functions reference these .rda files when defining the dictionary.

  • Input: year (2021, 2022, 2023, 2024)
  • Return: data/vars_yyyy.rda (as list)

Function:

getPUMSVars <- function(year){
  
  yearc <- as.character(year)

  ## Define variables of interest
  if (yearc %in% c("2021", "2022")) {
    want <- all_vars[-14]
  } else if (yearc %in% c("2023", "2024")) {
    want <- all_vars[-13]
  } else {
    stop("Invalid year input. Function only accepts 2021-2024")
  }
  
  ## Pull + Parse JSON
  url <- paste0("https://api.census.gov/data/",yearc,"/acs/acs1/pums/variables.json")
  pulled  <- httr::GET(url)
  parsed  <- fromJSON(rawToChar(pulled$content))
  
  ## Subset List to Variables of Interest
  vars <- parsed$variables[want]
  
  ## Save to /data as .rds
  saveRDS(vars, file= paste0("data/vars",yearc,".rds"))
}

API Pull: Helpers

The following helper functions perform dedicated tasks and are called within the main, wrapper function (getPUMSData defined later).

queryAPI()

This function queries the Census API for PUMAS data and returns a matrix of the requested data.

  • Input: all query parameters, as given to the main/wrapper function.
  • Return: a matrix
queryAPI <- function(year, num, cat, geo, subgeo,...){
  
  q <- paste(cat, "PWGTP", num, sep=",") 

  url <- paste0("https://api.census.gov/data/",as.character(year),
                "/acs/acs1/pums?get=",q,
                "&for=",tolower(geo),":",subgeo,
                "&SCHL=24&key=", my_key)
  
  pulled <- httr::GET(url)
  parsed <- fromJSON(rawToChar(pulled$content))
  
  ##REMOVE SCHL=24 later

}

makeTidy()

This function converts the requested data to a tibble with meaningful column names.

  • Input: a matrix generated by queryAPI
  • Return: a tibble
makeTidy <- function(parsed,...){
  
  #Get colnames from 1st row
  colnames(parsed) <- parsed[1,]
  
  #Make parsed tidy with colnames
  tdata  <- as_tibble(parsed) |> slice(-1) |>
    mutate(PWGTP=as.numeric(PWGTP))
}

cleanCat()

This function converts the requested categorical variable to a factor, based on the relevant data dictionary stored in /data.

  • Input: a tibble generated by makeTidy, the data dictionary, and the cat user input
  • Return: an updated tibble
cleanCat <- function(tdata,dictionary,cat,...){
  
  #Store variable labels
  lookup <- unlist(dictionary[[cat]]$values$item)
  
  #Apply labels and levels
  if(cat=="SEX") {
    tdata <- tdata |> mutate(SEX = factor(SEX, levels=names(lookup), labels=lookup))
  } else if(cat=="FER"){
    tdata <- tdata |> mutate(FER = factor(FER, levels=names(lookup), labels=lookup))
  } else if(cat=="HHL"){
    tdata <- tdata |> mutate(HHL = factor(HHL, levels=names(lookup), labels=lookup))
  } else if(cat=="SCH"){
    tdata <- tdata |> mutate(SCH = factor(SCH, levels=names(lookup), labels=lookup))
  } else if(cat=="SCHL"){
    tdata <- tdata |> mutate(SCHL = factor(SCHL, levels=names(lookup), labels=lookup))
  } 

}

cleanNum()

This function converts the requested numeric variable to an interpretable numeric value, based on the relevant data dictionary stored in /data.

Coded “N/A” responses are converted to NA elements, and time periods (JWAP and JWDP) are converted to midpoint time. Midpoint time is stored as R numeric data with the 12-hour time format (I:M p).

  • Input: a tibble generated by makeTidy, the data dictionary, and the num user input
  • Return: an updated tibble
cleanNum <- function(tdata,dictionary,num,...){
  
  #Store variable labels
  lookup <- unlist(dictionary[[num]]$values$item)
  
  #Apply labels and levels
  if(num=="AGEP") {
    tdata <- tdata |> mutate(AGEP = as.numeric(AGEP))
  } else if(num=="GASP"){
    tdata <- tdata |> mutate(GASP = as.numeric(GASP), 
                             GASP = ifelse(GASP==3,NA,GASP)) #set coded missing to zero
  } else if(num=="GRPIP"){
    tdata <- tdata |> mutate(GRPIP = as.numeric(GRPIP))
  } else if(num=="JWAP"){
    tdata <- tdata |> 
              #Convert time codes to character strings, use NA for coded "N/A"
              mutate(JWAPF = factor(JWAP, levels=names(lookup), labels=lookup),
                     JWAPC = ifelse(str_detect(JWAPF,"N/A"), NA, as.character(JWAPF))) |>
              #Parse time windows into start (t1) and stop (t2) times, all character values
              separate_wider_delim(cols=JWAPC, delim=" to ", names=c("t1","t2")) |>
              #Convert character time to numeric time and calculate midpoint time
              mutate(t1 = lubridate::parse_date_time(gsub("\\.", "", t1), orders = "I:M p"),
                     t2 = lubridate::parse_date_time(gsub("\\.", "", t2), orders = "I:M p"),
                     #replace original codes with midpoint:
                     JWAP = format(t1 + (t2-t1)/2, "%I:%M %p"))  |>
              select(-c(JWAPF,t1,t2))
  } else if(num=="JWDP"){
    tdata <- tdata |> 
              #Convert time codes to character strings, use NA for coded "N/A"
              mutate(JWDPF = factor(JWDP, levels=names(lookup), labels=lookup),
                     JWDPC = ifelse(str_detect(JWDPF,"N/A"), NA, as.character(JWDPF))) |>
              #Parse time windows into start (t1) and stop (t2) times, all character values
              separate_wider_delim(cols=JWDPC, delim=" to ", names=c("t1","t2")) |>
              #Convert character time to numeric time and calculate midpoint time
              mutate(t1 = lubridate::parse_date_time(gsub("\\.", "", t1), orders = "I:M p"),
                     t2 = lubridate::parse_date_time(gsub("\\.", "", t2), orders = "I:M p"),
                     #replace original codes with midpoint:
                     JWDP = format(t1 + (t2-t1)/2, "%I:%M %p")) |>
              select(-c(JWDPF,t1,t2))
  } else if(num=="JWMNP") {
    tdata <- tdata |> mutate(JWMNP = as.numeric(JWMNP))
  } 

}

cleanGeo()

This function converts the requested geographic variable to a factor based on the relevant data dictionary stored in /data.

  • Input: a tibble generated by makeTidy, the data dictionary, and the geo user input
  • Return: an updated tibble
cleanGeo <- function(tdata,dictionary,geo,year,...){
  
  #Store variable labels
  geod <- ifelse(geo=="STATE" & year %in% c(2021,2022), "ST", geo)
  lookup <- unlist(dictionary[[geod]]$values$item)
  
  #Apply labels and levels
  if(geo %in% c("STATE", "ST")) {
    tdata <- tdata |> mutate(state = factor(state, levels=names(lookup), labels=lookup))
  } else if(geo=="REGION"){
    tdata <- tdata |> mutate(region = factor(region, levels=names(lookup), labels=lookup))
  } else if(geo=="DIVISION"){
    tdata <- tdata |> mutate(division = factor(division, levels=names(lookup), labels=lookup))
  }

}

getOneYear()

This is a helper function that runs all prior helper functions. It is designed to query, tidy, and clean PUMS data for a single year. Within the main wrapper function (getPUMSData defined later), this helper function may be called multiple times to compile multiple years of data.

  • Input: all query parameters, as given to the main/wrapper function.
  • Return: a cleaned, single year tibble
getOneYear <- function(year, num, cat, geo, subgeo,...){

  ## CHECK SUBGEO AGAINST DICTIONARY
  dictionary <- readRDS(paste0("data/vars",as.character(year),".rds"))
  
  ##-----more to add
  
  ## RUN API PULL FUNCTION
  parsed <- queryAPI(year, num, cat, geo, subgeo)

  ## RUN TIDY FUNCTION
  tdata  <- makeTidy(parsed)
  
  ## RUN CLEANING FUNCTIONS
  tdata <- cleanCat(tdata,dictionary,cat)
  tdata <- cleanNum(tdata,dictionary,num)
  tdata <- cleanGeo(tdata,dictionary,geo,year)
  
  #Add year as a variable
  tdata <- tdata |> mutate(year=as.numeric(year))
  
  #Assign class
  class(tdata) <- c("census", class(tdata))
  return(tdata)
  
}

API Pull: Main

The getPUMSData function …

  • year accepts 2021, 2022, 2023, 2024
  • nums accepts one numeric variable: AGEP (default), GASP, GRPIP, JWAP, JWDP, JWMNP
    • Note: “PWGTP” is automatically included in addition to the requested variable
  • cats accepts one categorical variable: FER, HHL, SCH, SCHL, SEX (default).
  • geo accepts one level of geography: REGION, DIVISION, STATE (default)
    • Note: either “STATE” or “ST” may be used to request state-level geography
  • subgeo accepts one sub-level of geography: NULL (default), ALL, or a numeric code
    • The numeric code lists are specific to each level of geography. Please see the online PUMAS variable documentation for a complete list.
    • If no sub-level of geography is provided, the following defaults are used:
      • REGION = 1 (“Northeast”)
      • DIVISION = 1 (“New England (Northeast region)”)
      • STATE = 11 (“District of Columbia/DC”)
getPUMSData <- function(years=c(2024), num="AGEP", cat="SEX", geo="STATE", subgeo=NULL,...){

  ## CHECK INPUTS
  
  #years:
  years <- as.numeric(years)
  
  if(any(!as.character(years) %in% c("2021", "2022", "2023", "2024"))){
    stop("Invalid YEAR input. Function only accepts 2021-2024.")    
  } else if (length(unique(years)) != length(years)){
    stop("Invalid YEAR input. Same year entered multiple times.")
  }
  
  #num:
  num <- toupper(num)

  if (length(num)>1) {
    stop("Invalid NUM input. Only 1 variable permitted - PWGTP is automatically included.")
  } else if(num=="PWGTP"){
    stop("Invalid NUM input. PWGTP is automatically included, a different numeric variable is required.")
  } else if(any(!num %in% all_nums)){
    stop("Invalid NUM input. The requested variable is not on the approved list.")
  }

  #cat:
  cat <-toupper(cat)
  
  if (length(cat)>1) {
    stop("Invalid CAT input. Only 1 variable permitted.")
  } else if(any(!cat %in% all_cats)){
    stop("Invalid CAT input. The requested variable is not on the approved list.")
  }

  #geo:
  geo <- toupper(geo)

  if (length(geo) > 1){
    stop("Invalid GEO input. Only 1 geography variable may be provided.")
  } else if (!geo %in% c("REGION", "DIVISION", "STATE", "ST")){
    stop("Invalid GEO input. The requested variable is not on the approved geography list.")
  } else{
    geo <- ifelse(geo=="ST", "STATE", geo) 
  }
  
  #subgeo:
  #these are top-level checks, getOneYear checks year-specific dictionary compliance
  if(is.null(subgeo)){  
    #Set defaults
    subgeo <- ifelse(geo=="REGION", "1",
                ifelse(geo=="DIVISION", "1",
                  ifelse(geo %in% c("STATE","ST"), "11", NULL) ))
  } else if(length(subgeo) > 1){
    stop("Invalid SUBGEO input. Specify 'all' or a single level of sub-geography.")
  } else if(toupper(subgeo) %in% c("ALL","*")){
    subgeo <- "*"
  } else{
      subgeo <- as.character(subgeo)
  }
  
  #print final parameters to console
  message("Run for: ", years, " with ", num, " & ", cat, " for ", geo, " at ", subgeo)
  
  
  ## PULL CLEAN DATA
  all_years <- getOneYear(years[1],num,cat,geo,subgeo)
  
  if(length(years)>1){
     for (i in 2:length(years)){
        all_years <- all_years |> rbind(getOneYear(years[i],num,cat,geo,subgeo))
     }
  }
  
  return(all_years)
 
}
